2

יצירת פרויקט חדש

  1. היכנסו ל־console.cloud.google.com. בכניסה הראשונה תתבקשו לאשר את תנאי השימוש.
  2. למעלה, ליד הלוגו, לחצו על בורר הפרויקטים ואז על New project.
  3. תנו שם (למשל chat-bot) ולחצו Create.
  4. ודאו שהפרויקט החדש הוא זה שמופיע למעלה בבורר. אם לא, בחרו בו.

☰›APIs & Services›Enabled APIs & services

  3. היכנסו לתוצאה הראשונה ולחצו על הכפתור הכחול Enable.

הדבקת הקוד

בקובץ קוד.gs מחקו את כל מה שכתוב (Ctrl+A ואז Delete), העתיקו את הקוד הבא והדביקו:

קוד.gs · 128 שורות
// ✏️ כאן משנים את המילה (או כמה מילים) שמפעילה את הבוט
const TRIGGER_WORD = 'פיצי';

// נבנה אוטומטית מהמילה: @פיצי, גם עם רווח אחרי ה-@
const TRIGGER_RE = new RegExp(
  '@\\s*' + TRIGGER_WORD.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&').replace(/\s+/g, '\\s+')
);

const CONFIG = {
  // נבחרת תשובה אקראית מהרשימה. אפשר להשאיר אחת בלבד
  REPLIES: [
    'משמרת הצניעות כאן לשרותכם, ספרו לנו מה הבעיה ומי הבעיה, ונשמח לעזור ולנקות תוכן לא ראוי.'
  ],

  INCLUDE_DMS_AND_GROUPS: false, // true = גם שיחות אישיות וקבוצות, לא רק מרחבים
  COOLDOWN_SEC: 20,              // מקסימום תשובה אחת למרחב בפרק הזמן הזה
  MAX_RUN_MS: 240000,            // הגבלת זמן ריצה לסריקה אחת
  KEEP_DONE: 100                 // כמה הודעות לזכור (מגבלת 9KB לערך ב-Properties)
};

function installBotTrigger() {
  removeBotTrigger();
  PropertiesService.getUserProperties().setProperties({
    lastScan: new Date().toISOString(),
    done: '[]'
  });
  ScriptApp.newTrigger('scanMentions').timeBased().everyMinutes(1).create();
  Logger.log('הבוט הופעל.');
}

function removeBotTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'scanMentions') ScriptApp.deleteTrigger(t);
  });
  Logger.log('הבוט נעצר.');
}

function status() {
  const on = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'scanMentions');
  const p = PropertiesService.getUserProperties();
  Logger.log('פעיל: ' + on + ' | סריקה אחרונה: ' + p.getProperty('lastScan') + ' | מילת הפעלה: @' + TRIGGER_WORD);
}

function authorize() { Logger.log(Chat.Spaces.list({ pageSize: 1 })); }

function scanMentions() {
  const lock = LockService.getUserLock();
  if (!lock.tryLock(5000)) return;
  try {
    const props = PropertiesService.getUserProperties();
    const started = new Date();
    const since = new Date(props.getProperty('lastScan') || started.toISOString());
    // חפיפה של דקה כדי לא לפספס הודעות; כפילויות נחסמות ע"י רשימת done
    const cut = new Date(since.getTime() - 60000).toISOString().replace(/\.\d+Z$/, '+00:00');
    const done = JSON.parse(props.getProperty('done') || '[]');
    let complete = true;

    for (const space of listSpaces_()) {
      if (new Date() - started > CONFIG.MAX_RUN_MS) { complete = false; break; }

      let msgs = [];
      try {
        msgs = Chat.Spaces.Messages.list(space, {
          pageSize: 50,
          filter: 'createTime > "' + cut + '"'
        }).messages || [];
      } catch (err) { continue; }

      for (const m of msgs) {
        if (done.indexOf(m.name) > -1) continue;
        if (!TRIGGER_RE.test(m.text || '')) continue;
        done.push(m.name);
        if (onCooldown_(space)) continue;
        try { botReply_(space, m); } catch (err) { Logger.log(err); }
      }
    }

    props.setProperties({
      lastScan: complete ? started.toISOString() : since.toISOString(),
      done: JSON.stringify(done.slice(-CONFIG.KEEP_DONE))
    });
  } finally {
    lock.releaseLock();
  }
}

function botReply_(space, m) {
  const text = CONFIG.REPLIES[Math.floor(Math.random() * CONFIG.REPLIES.length)];
  const msg = {
    text: text,
    quotedMessageMetadata: { name: m.name, lastUpdateTime: m.lastUpdateTime || m.createTime }
  };
  if (m.thread && m.thread.name) msg.thread = { name: m.thread.name };
  try {
    Chat.Spaces.Messages.create(msg, space, { messageReplyOption: 'REPLY_MESSAGE_FALLBACK_TO_NEW_THREAD' });
  } catch (err) {
    // אם הציטוט נכשל, שולחים הודעה רגילה
    Chat.Spaces.Messages.create({ text: text }, space);
  }
}

// רשימת כל המרחבים (עם דפדוף ו-cache של 5 דקות כדי לחסוך קריאות)
function listSpaces_() {
  const cache = CacheService.getUserCache();
  const hit = cache.get('spaces');
  if (hit) return JSON.parse(hit);

  const names = [];
  let token;
  do {
    const opt = { pageSize: 200, pageToken: token };
    if (!CONFIG.INCLUDE_DMS_AND_GROUPS) opt.filter = 'spaceType = "SPACE"';
    const res = Chat.Spaces.list(opt);
    (res.spaces || []).forEach(s => names.push(s.name));
    token = res.nextPageToken;
  } while (token);

  try { cache.put('spaces', JSON.stringify(names), 300); } catch (err) {}
  return names;
}

function onCooldown_(space) {
  const cache = CacheService.getUserCache();
  const key = 'cd:' + space;
  if (cache.get(key)) return true;
  cache.put(key, '1', Math.max(CONFIG.COOLDOWN_SEC, 1));
  return false;
}

השורות המסומנות הן אלה שמשנים (ראו התאמה אישית). בינתיים אפשר להשאיר כמו שזה.

ההעתקה האוטומטית לא עבדה. הקוד סומן: לחצו Ctrl+C.

בדיקה

כתבו באחד המרחבים שאתם חברים בו @פיצי (או המילה שבחרתם). תוך דקה–שתיים אמורה להגיע תשובה עם ציטוט של ההודעה.

טיפ: כיוון ש"פיצי" הוא לא משתמש אמיתי, הצ'אט לא יציע אותו בהשלמה אוטומטית. פשוט מקלידים את הטקסט. מותר גם רווח אחרי ה־@.

התאמה אישית

כל ההגדרות נמצאות בראש הקובץ קוד.gs.

הגדרהמה היא עושה
TRIGGER_WORDמילת ההפעלה, בלי ה־@. אפשר גם כמה מילים, למשל 'משמרת הצניעות'.
REPLIESהתשובה. אם כותבים כמה תשובות, בכל פעם נבחרת אחת באקראי.
COOLDOWN_SECלכל היותר תשובה אחת למרחב בפרק הזמן הזה (בשניות). ברירת מחדל: 20.
INCLUDE_DMS_AND_GROUPSfalse: עונה רק במרחבים. true: עונה גם בצ'אטים אישיים ובקבוצות.

אחרי כל שינוי שומרים עם Ctrl+S, וזהו. לא צריך להפעיל את הבוט מחדש: הסריקה הבאה כבר משתמשת בקוד החדש.

תשובות חכמות עם בינה מלאכותית

במקום תשובה קבועה, הבוט שולח את ההודעה ל־Gemini (הבינה המלאכותית של Google) ועונה במה שהוא ניסח. הכול קורה בתוך המרחבים הקיימים, בלי להוסיף אפליקציה לצ'אט.

לא עובד בנטפרי. הקוד פונה לשירות חיצוני (Gemini), ולפי מון בלאן נטפרי חוסם את זה. השדרוג מיועד למי שלא בנטפרי.

קודם משלימים את שלבים 1–10 ומוודאים שהגרסה הרגילה עובדת. אחר כך:

א

  1. היכנסו ל־aistudio.google.com/apikey ולחצו Create API key.

השימוש חינמי עד מגבלה יומית. Google משנה את המגבלות מדי פעם, והמספרים המעודכנים מופיעים ב־AI Studio.

ב

שמירת המפתח בסקריפט

  1. ב־⚙️ הגדרות הפרויקט גללו לתחתית, לחלק מאפייני סקריפט (Script Properties).
  2. לחצו הוספת מאפיין סקריפט.
  3. בשדה המאפיין (Property) כתבו
  4. בשדה הערך (Value) הדביקו את המפתח.
  5. לחצו שמירת מאפייני הסקריפט.

ככה המפתח לא כתוב בתוך הקוד, ואפשר לשתף את הקוד בלי לחשוף אותו.

// 🤖 גרסת בינה מלאכותית (Gemini). לא עובדת בנטפרי.
// ✏️ כאן משנים את המילה (או כמה מילים) שמפעילה את הבוט
const TRIGGER_WORD = 'פיצי';

// נבנה אוטומטית מהמילה: @פיצי, גם עם רווח אחרי ה-@
const TRIGGER_RE = new RegExp(
  '@\\s*' + TRIGGER_WORD.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&').replace(/\s+/g, '\\s+')
);

const CONFIG = {
  // ✏️ ההנחיה לבינה המלאכותית: מי היא ואיך לענות
  AI_INSTRUCTIONS: 'אתה עוזר בקבוצת גוגל צ׳אט. ענה בעברית, בקצרה ובנימוס, בטקסט רגיל בלי כוכביות ובלי עיצוב Markdown.',

    'סליחה, לא הצלחתי לענות כרגע. נסו שוב בעוד כמה דקות.'
  ],

  INCLUDE_DMS_AND_GROUPS: false, // true = גם שיחות אישיות וקבוצות, לא רק מרחבים
  COOLDOWN_SEC: 20,              // מקסימום תשובה אחת למרחב בפרק הזמן הזה
  MAX_RUN_MS: 240000,            // הגבלת זמן ריצה לסריקה אחת
  KEEP_DONE: 100                 // כמה הודעות לזכור (מגבלת 9KB לערך ב-Properties)
};

function installBotTrigger() {
  removeBotTrigger();
  PropertiesService.getUserProperties().setProperties({
    lastScan: new Date().toISOString(),
    done: '[]'
  });
  ScriptApp.newTrigger('scanMentions').timeBased().everyMinutes(1).create();
  Logger.log('הבוט הופעל.');
}

function removeBotTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'scanMentions') ScriptApp.deleteTrigger(t);
  });
  Logger.log('הבוט נעצר.');
}

}

// בדיקה: שולח ל-Gemini שאלה לדוגמה ומציג את התשובה ביומן
function testAI() {
  Logger.log(aiReply_({ text: '@' + TRIGGER_WORD + ' שלום! מה אתה יודע לעשות?' }) || 'לא התקבלה תשובה. בדקו את המפתח ואת השגיאה ביומן.');
}

function scanMentions() {
  const lock = LockService.getUserLock();
  if (!lock.tryLock(5000)) return;
  try {
    const props = PropertiesService.getUserProperties();
    const started = new Date();
    const since = new Date(props.getProperty('lastScan') || started.toISOString());
    // חפיפה של דקה כדי לא לפספס הודעות; כפילויות נחסמות ע"י רשימת done
    const cut = new Date(since.getTime() - 60000).toISOString().replace(/\.\d+Z$/, '+00:00');
    const done = JSON.parse(props.getProperty('done') || '[]');
    let complete = true;

    for (const space of listSpaces_()) {
      if (new Date() - started > CONFIG.MAX_RUN_MS) { complete = false; break; }

      let msgs = [];
      try {
        msgs = Chat.Spaces.Messages.list(space, {
          pageSize: 50,
          filter: 'createTime > "' + cut + '"'
        }).messages || [];
      } catch (err) { continue; }

      for (const m of msgs) {
        if (done.indexOf(m.name) > -1) continue;
        if (!TRIGGER_RE.test(m.text || '')) continue;
        done.push(m.name);
        if (onCooldown_(space)) continue;
        try { botReply_(space, m); } catch (err) { Logger.log(err); }
      }
    }

    props.setProperties({
      lastScan: complete ? started.toISOString() : since.toISOString(),
      done: JSON.stringify(done.slice(-CONFIG.KEEP_DONE))
    });
  } finally {
    lock.releaseLock();
  }
}

function botReply_(space, m) {
  const text = aiReply_(m) || CONFIG.REPLIES[Math.floor(Math.random() * CONFIG.REPLIES.length)];
  const msg = {
    text: text,
    quotedMessageMetadata: { name: m.name, lastUpdateTime: m.lastUpdateTime || m.createTime }
  };
  if (m.thread && m.thread.name) msg.thread = { name: m.thread.name };
  try {
    Chat.Spaces.Messages.create(msg, space, { messageReplyOption: 'REPLY_MESSAGE_FALLBACK_TO_NEW_THREAD' });
  } catch (err) {
    // אם הציטוט נכשל, שולחים הודעה רגילה
    Chat.Spaces.Messages.create({ text: text }, space);
  }
}

// שולח את ההודעה ל-Gemini ומחזיר את התשובה, או null אם משהו נכשל
function aiReply_(m) {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) { Logger.log('חסר GEMINI_API_KEY במאפייני הסקריפט.'); return null; }

  // ההודעה בלי התיוג עצמו
  const tagAll = new RegExp(TRIGGER_RE.source, 'g');
  const question = (m.text || '').replace(tagAll, ' ').trim() || 'שלום';

  try {
    const res = UrlFetchApp.fetch(
      'https://generativelanguage.googleapis.com/v1beta/models/' + CONFIG.AI_MODEL + ':generateContent',
      {
        method: 'post',
        contentType: 'application/json',
        headers: { 'x-goog-api-key': key },
        payload: JSON.stringify({
          systemInstruction: { parts: [{ text: CONFIG.AI_INSTRUCTIONS }] },
          contents: [{ role: 'user', parts: [{ text: question }] }]
        }),
        muteHttpExceptions: true
      }
    );
    if (res.getResponseCode() !== 200) {
      Logger.log('Gemini החזיר שגיאה ' + res.getResponseCode() + ': ' + res.getContentText().slice(0, 300));
      return null;
    }
    const data = JSON.parse(res.getContentText());
    const c = (data.candidates || [])[0];
    const parts = (c && c.content && c.content.parts) || [];
    let out = parts.filter(p => p.text && !p.thought).map(p => p.text).join('').trim();
    // שהתשובה לא תכיל את התיוג, אחרת הבוט יענה לעצמו בלולאה
    out = out.replace(tagAll, TRIGGER_WORD);
    return out ? out.slice(0, 4000) : null;
  } catch (err) {
    Logger.log('Gemini: ' + err);
    return null;
  }
}

// רשימת כל המרחבים (עם דפדוף ו-cache של 5 דקות כדי לחסוך קריאות)
function listSpaces_() {
  const cache = CacheService.getUserCache();
  const hit = cache.get('spaces');
  if (hit) return JSON.parse(hit);

  const names = [];
  let token;
  do {
    const opt = { pageSize: 200, pageToken: token };
    if (!CONFIG.INCLUDE_DMS_AND_GROUPS) opt.filter = 'spaceType = "SPACE"';
    const res = Chat.Spaces.list(opt);
    (res.spaces || []).forEach(s => names.push(s.name));
    token = res.nextPageToken;
  } while (token);

  try { cache.put('spaces', JSON.stringify(names), 300); } catch (err) {}
  return names;
}

function onCooldown_(space) {
  const cache = CacheService.getUserCache();
  const key = 'cd:' + space;
  if (cache.get(key)) return true;
  cache.put(key, '1', Math.max(CONFIG.COOLDOWN_SEC, 1));
  return false;
}

  • יש עיכוב של עד כדקה, כי הסריקה רצה פעם בדקה.
  • מגבלה יומית: בחשבון Gmail רגיל יש ל־Google מכסה יומית לזמן הריצה של סקריפטים אוטומטיים (בערך 90 דקות ביום). ככל שאתם חברים ביותר מרחבים, כל סריקה ארוכה יותר. אם הבוט מפסיק לענות לקראת סוף היום, כנראה הגעתם למכסה, והוא יחזור לעבוד כשהיא תתאפס.
  • נטפרי: לדברי מון בלאן הקוד עובד גם בנטפרי, כי הוא לא פונה לשירות חיצוני. חיבור לבינה מלאכותית (כמו Gemini) כן פונה לשירות חיצוני, ולכן נחסם.
